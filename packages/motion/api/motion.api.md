/**
 * Caption paging: word-level `captions.json` into the pages the caption track shows one at a time.
 *
 * **The rule, which `packages/render-core` already writes `captions.json` against.** Its
 * `buildCaptions()` gives every word but the first of the whole track a leading space and never
 * gives one to folded punctuation (`packages/render-core/src/narrate/captions.ts`). The pager is
 * the other half of that agreement:
 *
 * 1. Captions are taken in order. The first one opens the first page.
 * 2. A caption **opens a new page only if its `text` begins with whitespace** and at least
 *    `combineWithinMs` have elapsed from the current page's start to the caption's `startMs`. The
 *    window is half-open: a caption exactly `combineWithinMs` after the page start opens the next
 *    page. A caption with no leading space never opens a page however late it is — it is the tail
 *    of the word before it, and splitting there would strand it.
 * 3. The caption that opens a page has its leading whitespace trimmed, in the page `text` and in
 *    its token, so no page begins with a space. Every other caption is appended verbatim: page
 *    `text` is the plain concatenation of its tokens' `text`.
 * 4. A page lasts until the next page starts; the last page lasts until the latest `endMs` among
 *    its own tokens. So consecutive pages tile the time between them with no gap for the caption
 *    band to flicker through, and the track ends when the speech does.
 *
 * Grouping is by elapsed time only; nothing here knows where a sentence or a narration segment
 * ends. The scaffold's `Captions.tsx` gets sentence-aware pages by calling this once per sentence,
 * which is why this function stays this small.
 *
 * Pure: no clock, no frame, no order dependence, so a frame rendered out of order or a still taken
 * alone pages exactly as the full render does.
 */
/** One word of `captions.json`, in the shape `packages/protocol/schemas/captions.json` defines. */
export type Caption = {
    text: string;
    startMs: number;
    endMs: number;
    timestampMs: number | null;
    confidence: number | null;
};

/** One caption inside a page, with its own span so the spoken word can be highlighted. */
export type TikTokToken = {
    text: string;
    fromMs: number;
    toMs: number;
};

/** One page of caption text and the tokens it was built from. */
export type TikTokPage = {
    text: string;
    startMs: number;
    durationMs: number;
    tokens: TikTokToken[];
};

/** The fields of a caption the pager reads. Every `Caption` is one. */
export type PageableCaption = Pick<Caption, "text" | "startMs" | "endMs">;

export type PaginateCaptionsOptions = {
    /** A caption starting this long after its page's start (or later) may open the next page. */
    combineWithinMs: number;
};

export declare function paginateCaptions(captions: readonly PageableCaption[], options: PaginateCaptionsOptions): TikTokPage[];

export type CreateTikTokStyleCaptionsInput = {
    captions: readonly PageableCaption[];
    combineTokensWithinMilliseconds: number;
};

/**
 * `paginateCaptions` under its older name and argument shape.
 *
 * It exists so scenes written before this package — which imported the pager from a
 * captions-specific module under this name — keep compiling through the renderer's import alias
 * for one release. New code should call `paginateCaptions`.
 */
export declare function createTikTokStyleCaptions(input: CreateTikTokStyleCaptionsInput): {
    pages: TikTokPage[];
};

## `dist/easing.d.ts`

```ts
/**
 * `Easing` — the curves a scene hands to `interpolate()` to shape a transition.
 *
 * **Every curve here is a documented formula, written from the formula.** The set and the
 * names follow React Native's `Easing` module, because that is the vocabulary explainer scenes
 * are already written in; the bodies are the textbook closed forms (Penner's equations and the
 * CSS cubic-Bézier definition), not a port of anybody's source. `packages/motion/src/math.test.ts`
 * holds each one against a separate evaluation of the same formula, so a curve that drifts is a
 * red test rather than a scene that looks slightly wrong.
 *
 * **The domain is not clamped.** `interpolate()` with `extrapolate*: "extend"` feeds a curve
 * positions below 0 and above 1, and every polynomial curve here simply continues. The one curve
 * that cannot continue on its own is `bezier`, whose parametric form is not a function outside
 * `[0, 1]`; it extends along the tangent at the nearer endpoint, which is what the CSS Easing
 * specification prescribes for the same situation.
 */
/** A curve over normalised time: 0 at the start of a segment, 1 at its end. */
export type EasingFunction = (t: number) => number;

/** The shape of `Easing`, written out because a const this public carries its type explicitly. */
export type EasingModule = {
    /** `t`. */
    readonly linear: EasingFunction;
    /** `bezier(0.42, 0, 1, 1)` — a gentle start. */
    readonly ease: EasingFunction;
    /** `t²`. */
    readonly quad: EasingFunction;
    /** `t³`. */
    readonly cubic: EasingFunction;
    /** `tⁿ`. */
    readonly poly: (n: number) => EasingFunction;
    /** `1 − cos(t·π/2)`. */
    readonly sin: EasingFunction;
    /** `1 − √(1 − t²)`. */
    readonly circle: EasingFunction;
    /** `2^(10·(t − 1))`. */
    readonly exp: EasingFunction;
    /** `1 − cos(t·π/2)³ · cos(t·bounciness·π)`; `bounciness` defaults to 1. */
    readonly elastic: (bounciness?: number) => EasingFunction;
    /** `t²·((s + 1)·t − s)`; `s` defaults to 1.70158, a 10 % overshoot. */
    readonly back: (s?: number) => EasingFunction;
    /** Penner's four-arc bounce. */
    readonly bounce: EasingFunction;
    /** The CSS `cubic-bezier(x1, y1, x2, y2)` curve. */
    readonly bezier: (x1: number, y1: number, x2: number, y2: number) => EasingFunction;
    /** The curve as given: the "in" half is the curve itself. */
    readonly in: (easing: EasingFunction) => EasingFunction;
    /** The curve mirrored in time and value: `1 − f(1 − t)`. */
    readonly out: (easing: EasingFunction) => EasingFunction;
    /** "in" for the first half, "out" for the second, meeting at (0.5, 0.5). */
    readonly inOut: (easing: EasingFunction) => EasingFunction;
    /** 0 at `t ≤ 0`, then 1. */
    readonly step0: EasingFunction;
    /** 0 until `t ≥ 1`, then 1. */
    readonly step1: EasingFunction;
};

export declare const Easing: EasingModule;
```

## `dist/random.d.ts`

```ts
/**
 * `random(seed)` — a deterministic number in `[0, 1)` per seed.
 *
 * A scene that scatters particles with `Math.random()` draws a different picture on every frame:
 * a render seeds `Math.random` once, so every run draws the same sequence, but each frame's render
 * takes the next values from it and the particles jump. Keyed by a seed, the same particle lands
 * in the same place on every frame, in every process.
 *
 * The seed is hashed with 32-bit FNV-1a over its UTF-16 code units (a number seed is hashed as
 * its decimal string, so `random(7)` equals `random("7")`) and the hash seeds one step of
 * mulberry32. Both are public-domain, well-studied and a few lines long. The values are stable
 * across releases of this package but are **not** the values any other library returns for the
 * same seed: a scene ported from elsewhere gets a different, equally deterministic, scatter.
 *
 * `null` opts out of the seed and returns `Math.random()`: in a render, the next value of the
 * sequence the harness seeds, which differs from frame to frame but not from run to run.
 */
export declare function random(seed: number | string | null): number;
```

## `dist/spring.d.ts`

```ts
/**
 * `spring()` and `measureSpring()` — physical motion as a pure function of the frame.
 *
 * **The curve is the closed-form solution of a damped harmonic oscillator**, not a step-by-step
 * simulation. A mass `m` on a spring of stiffness `k` with damping `c` starts at rest at 0 and is
 * released towards 1:
 *
 *     m·x″ + c·x′ + k·(x − 1) = 0,   x(0) = 0,   x′(0) = 0
 *
 * With `ω₀ = √(k/m)` and `ζ = c / (2·√(k·m))` the textbook solution has three regimes —
 * under-damped (`ζ < 1`, it overshoots and rings), critically damped (`ζ = 1`) and over-damped
 * (`ζ > 1`, it creeps in without overshooting) — and each is evaluated directly at the frame's
 * time. Two consequences matter for a renderer that captures frames in any order: the value at
 * frame 90 does not depend on frames 0–89 having been computed first, and it is identical whether
 * the video is rendered or a single still is taken. `math.test.ts` holds the curve against an
 * independent RK4 integration of the equation above at 1e-9.
 *
 * The curve is normalised (0 → 1) and then scaled onto `from → to`, so `overshootClamping`, the
 * rest threshold and the duration stretch all mean the same thing whatever the two ends are.
 *
 * **`overshootClamping` stops the motion at the target.** It is React Native Animated's flag of
 * the same name: once the value first reaches `to`, it stays there. That is the first root of
 * `x(t) = 1`, at `ω_d·t* = π − atan2(ω_d, ζ·ω₀)`, and only an under-damped spring has one —
 * critically and over-damped motion never reaches the target, so the flag changes nothing for them.
 * Clamping each sample to `min(x, 1)` instead would not be the same thing: after the first swing
 * the oscillator falls back below the target, so a clamped sample sinks away from `to` and climbs
 * back — the default spring at 30 fps would hold 1.000 from frame 8 to 18, dip to 0.973 at 22 and
 * return at 29.
 *
 * The values follow the equation, not any other library's numerical scheme, so a scene ported
 * from elsewhere can differ in the last few hundredths of its overshoot.
 */
export type SpringConfig = {
    /** Default 1. */
    mass: number;
    /** Default 10. 0 never settles; `measureSpring` refuses it. */
    damping: number;
    /** Default 100. */
    stiffness: number;
    /** Default false. When true the value stops at `to` the first time it reaches it. */
    overshootClamping: boolean;
};

export type SpringOptions = {
    /** The frame to evaluate, usually `useCurrentFrame()`. */
    frame: number;
    /** Frames per second, usually `useVideoConfig().fps`. */
    fps: number;
    config?: Partial<SpringConfig>;
    /** Default 0. */
    from?: number;
    /** Default 1. */
    to?: number;
    /** Stretch time so the spring's natural settle (`measureSpring`) lands on this many frames. */
    durationInFrames?: number;
    /** Frames to hold at the start value before the spring is released. Default 0. */
    delay?: number;
    /** Run the motion backwards in time: settled at the start, back at `from` at the end. */
    reverse?: boolean;
    /** The threshold `durationInFrames` and `reverse` measure the natural settle with. Default 0.005. */
    durationRestThreshold?: number;
};

export type MeasureSpringOptions = {
    fps: number;
    config?: Partial<SpringConfig>;
    /** How close to the target, in normalised units (0 → 1), counts as settled. Default 0.005. */
    threshold?: number;
};

/**
 * The number of frames until the spring stays within `threshold` of its target for good: the
 * first frame `F` such that every frame from `F` on is inside. A spring with no damping never
 * settles, and is refused rather than looped over; so is one that takes more than an hour to
 * settle — so lightly damped it rings for that long, or so heavily that it creeps — with a
 * `RangeError` that says which.
 */
export declare function measureSpring(options: MeasureSpringOptions): number;

export declare function spring(options: SpringOptions): number;
```

/**
 * `staticFile(name)` — the URL of a file the narration step wrote beside the video.
 *
 * A scene never knows where its bundle will be opened from: the renderer writes it to a per-job
 * directory and serves it from a loopback address on a port of its own. So the base is not baked
 * into the scene; the bundle's
 * classic `static-data.js` sets `globalThis.__XPLAINER_PUBLIC_BASE__` (a URL ending in `/`, pointing
 * at `public/<slug>/`) before any module runs, and this function appends the name to it. In Node,
 * where nothing sets it, the base is empty and the result is the encoded name alone — which is
 * what makes this function testable without a page.
 *
 * Each path segment is `encodeURIComponent`-encoded and the slashes between them are kept, so
 * `staticFile("shots/step 1.png")` is `<base>shots/step%201.png`. A name that is already encoded is
 * encoded again; pass the name as it is on disk.
 *
 * **The name stays inside `public/<slug>/`, or it is refused.** A URL is refused rather than
 * mangled, and so are the two spellings of a path that would resolve somewhere else: one starting
 * with `/`, and one with a `..` segment, which the browser resolves against the base and walks out
 * of the video's own directory. Each is a `TypeError` naming the name and the rule.
 */
export declare function staticFile(name: string): string;
