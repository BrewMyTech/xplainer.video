## `dist/contract-version.d.ts`

```ts
/**
 * Whether a shim speaking `shim` may attach to a daemon speaking `daemon`.
 *
 * True when both versions parse and share a major component. The relation is
 * **symmetric** on purpose: a newer shim meeting an older daemon is the upgrade
 * case, and an older shim meeting a newer daemon is the not-yet-restarted-agent
 * case, and neither is more dangerous than the other once unknown enum members
 * decode instead of throwing.
 *
 * A version this build cannot parse is **incompatible**, never "probably fine".
 * That is the fail-closed direction: the cost of refusing is one legible error
 * naming both versions, and the cost of attaching anyway is the illegible
 * failure ADR 0025 was written to eliminate.
 *
 * @param daemon The contract version the daemon advertised, from `/healthz`'s
 *   `contract_version` or the ready line's `contract`.
 * @param shim This process's own {@link MCP_CONTRACT_VERSION}.
 * @returns `true` if the pair may speak to each other.
 */
export declare function isContractCompatible(daemon: string, shim: string): boolean;
```

export type ToolName = (typeof TOOL_NAMES)[number];

/**
 * The scaffold files the engine owns, in scaffold order.
 *
 * These five mount the narration audio, the caption track and the per-segment
 * sequencing. An agent that rewrote one could drop the soundtrack and still
 * render successfully, so `explainer_put_source` refuses to write them and the
 * scaffold generator restores them from its templates. The agent owns
 * `Scenes.tsx` and everything it imports instead.

 * lives in `schemas/manifest.json`, and no consumer gets to extend or narrow it
 * at runtime.
 */
export declare const ENGINE_OWNED_FILES: readonly ["index.ts", "types.ts", "Root.tsx", "Captions.tsx", "Video.tsx"];

/** One of the engine-owned scaffold file names. */
export type EngineOwnedFile = (typeof ENGINE_OWNED_FILES)[number];

/**
 * The version of the **tool contract**, from `schemas/manifest.json`.
 *
 * Not a release version. `@xplainer/cli` and the hosted image each have their
 * own `version`, and the MCP `initialize` handshake reports *that* in
 * `serverInfo.version`, which is why the handshake cannot be used to detect
 * contract skew. This number is what `xplainer serve` advertises on
 * `GET /healthz` as `contract_version` and what
 * {@link isContractCompatible} compares
 * ([ADR 0025](../../../../docs/adr/0025-daemon-updates-and-readiness.md)
 * §Note, 2026-09-06: P1-S3 settled).
 *
 * It lives here, in the package that owns the contract, rather than in
 * `@xplainer/mcp-server`: the `xplainer mcp --attach` shim must read it
 * before any MCP session exists, and `@xplainer/protocol` is the one package
 * every surface already depends on.
 */
export declare const MCP_CONTRACT_VERSION: string;

export interface Caption {
    /**
     * The word. Whitespace-sensitive: every token except the first of the whole track carries its leading space, so pages join correctly. A segment boundary is not a page boundary, so the first word of a segment carries one too.
     */
    text: string;
    /**
     * Word start on the narration track, in milliseconds.
     */
    startMs: number;
    /**
     * Word end on the narration track, in milliseconds.
     */
    endMs: number;
    /**
     * Midpoint of the word, which is what @remotion/captions pages on.
     */
    timestampMs: number;
    /**
     * Recogniser confidence. Always null here: Kokoro synthesises the audio, so the timings are exact rather than estimated.
     */
    confidence: number | null;
}

/**
 * Word-level captions in @remotion/captions Caption[] form, written to captions.json. Emitted by max/.explainers/scripts/narrate.py:244-253 from Kokoro's word timestamps, so caption timing is measured rather than inferred.

export type Captions = Caption[];

/**
 * Machine-readable reason a job ended in error, which an agent can branch on. It never replaces `error`: that field always carries the human-readable detail, and this one only says which class of failure produced it. `daemon_restarted`: the daemon restarted while the job was in flight, so the work was lost rather than rejected — retry the same call. `daemon_shutdown`: the daemon stopped before the job finished — retry once it is running again. `toolchain_missing`: a required binary is not installed — do not retry until it is, because every attempt fails the same way. `render_failed`: the render ran and failed — read `error`, fix the source, then retry. `tts_failed`: narration synthesis failed — retry once, and treat a second identical failure as a problem with the narration rather than the run. `cancelled`: a caller cancelled the job — do not retry unless the caller asks again. `internal`: an unclassified failure — retry once, and report `error` verbatim if it repeats. This enum is expected to grow as failure modes are told apart, and it is OPEN: adding a member is a minor contract change, and a consumer that meets a value it does not recognise falls back to `internal` instead of rejecting the record. Both generated bindings already do that — `toJobErrorCode()` in TypeScript and the `_missing_` hook on the Python enum — so branch on the decoded value and quote `error` for the detail. What adding a member costs is settled in ADR 0024's note of 2026-09-06.
 */
export type JobErrorCode = "daemon_restarted" | "daemon_shutdown" | "toolchain_missing" | "render_failed" | "tts_failed" | "cancelled" | "internal";

/**
 * The tail of the job's console output, oldest line first.
 */
export interface JobOutput {
    /**
     * At most output_lines lines, in the order they were written.
     */
    lines: string[];
}

/**
 * Lifecycle state of a queued explainer job. The five values are the ones the reference implementation's job queue reports and are the complete set: a job is always in exactly one of them.
 */
export type JobState = "queued" | "running" | "done" | "error" | "cancelled";

/**
 * Which tool queued the job. Only these three enqueue work.
 */
export type JobType = "explainer_narrate" | "explainer_render" | "explainer_still";

/**
 * The narration spec an agent hands to explainer_narrate. Field defaults mirror max/.explainers/scripts/narrate.py:191-196, which is the implementation that consumes this document.
 */
export interface Narration {
    /**
     * Kokoro voice id, or a blend such as "af_bella(2)+af_sky(1)". Blends are resolved server-side.
     */
    voice?: string;
    /**
     * Speech rate multiplier passed straight through to Kokoro.
     */
    speed?: number;
    /**
     * Frame rate the measured timings are converted to. Must match the composition.
     */
    fps?: number;
    /**
     * Optional Kokoro language code. Inferred from the voice when omitted.
     */
    langCode?: string;
    /**
     * Ordered narration segments. One timings entry is produced per segment, in this order.
     *
     * @minItems 1
     */

export interface NarrationSegment {
    /**
     * Stable scene identifier. Reappears as the id of the matching timings segment.
     */
    id: string;
    /**
     * What the voice says. Empty text yields a silent segment of holdSeconds length.
     */
    text: string;
    /**
     * Minimum on-screen duration. Silence is appended when the narration is shorter, so an animation can finish or a beat can land.
     */
    holdSeconds?: number;
}

/**
 * Identifier for one explainer video. Lowercase letters, digits and hyphens; must start with a letter or digit; at most 64 characters. Ported verbatim from the reference implementation's SLUG_RE, because the slug is also a directory name on every backend.
 */
export type Slug = string;

export interface SourceFile {
    /**
     * A relative path built from plain segments. No leading slash, and no segment may begin with a dot, so ".." cannot appear and a write can never escape the video directory. The pattern is deliberately lookahead-free: pydantic v2 compiles it with the Rust regex engine, which has no lookaround. The `not`/`enum` reserves the five engine-owned files listed in manifest.json's engine_owned_files: index.ts, types.ts, Root.tsx, Captions.tsx and Video.tsx mount the narration audio, the caption track and the per-segment sequencing, so an agent that rewrote one could ship a silent video. Only those exact top-level paths are reserved — a nested scenes/Root.tsx is the agent's and stays writable. `not`/`enum` rather than a negative lookahead for the same Rust-regex reason. Both code generators ignore `not`, so this is the published contract and the ajv gate, not the runtime gate; the runtime gate is assertAgentOwnedPaths() in @xplainer/mcp-server.
     */
    path: string;
    /**
     * UTF-8 file contents.
     */
    content: string;
}

/**
 * Measured segment timing, written to timings.json beside the narration audio. This document — never a hand-written number — is the single source of truth for scene durations. Shape mirrors the reference implementation's timings type and its narration writer.

    /**

    /**
     * Total narration track length in milliseconds.
     */
    totalMs: number;
    /**
     * Filename of the narration track, relative to the video's public directory.
     */
    audio: string;
    /**
     * One entry per narration segment, in narration order.
     */
    segments: TimingsSegment[];
}

export interface TimingsSegment {
    /**
     * The id of the narration segment this was measured from.
     */
    id: string;
    /**
     * Zero-based position in the narration.
     */
    index: number;
    /**
     * Segment start on the narration track, in milliseconds.
     */
    startMs: number;
    /**
     * Segment end on the narration track, in milliseconds.
     */
    endMs: number;
    /**
     * Segment start as a frame number, for Remotion's Sequence.

    /**
     * Segment length in frames. Never below 1.

     */
    seconds?: number;
